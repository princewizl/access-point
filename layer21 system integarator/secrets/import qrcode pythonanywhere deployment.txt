import qrcode

# Example payload from FIRS spec
payload = "TIN=1234567890;InvoiceNo=INV001;Date=2025-08-17;Amount=5000;..."

qr = qrcode.QRCode(
    version=1,
    error_correction=qrcode.constants.ERROR_CORRECT_Q,
    box_size=10,
    border=4,
)
qr.add_data(payload)
qr.make(fit=True)

img = qr.make_image(fill_color="black", back_color="white")
img.save("invoice_qr.png")
